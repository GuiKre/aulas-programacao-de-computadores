1 - Escreva uma função que retorne o valor absoluto de um número.

#include <stdio.h>
 
int absoluto(int num){
    
	if(num >= 0){
		return num;
	}
	else{
		return num * -1;
	}
}
 
int main() {
    
	int num;
    
	printf("Informe o número: ");
	scanf("%d", &num);
    
	printf("O valor absoluto é: %d", absoluto(num));
    
	return 0;
}

3 - Escreva uma função que retorne o maior valor entre três números inteiros.

#include <stdio.h>

	printf("O maior número é: %d", maior(num1, num2, num3));

}


4 - Escreva uma função que receba uma letra e 3 notas de um aluno. Se a letra for A, a função deve retornar a média aritmética das notas do aluno e se a letra for P, a função deve retornar a sua média ponderada com os pesos 5, 3 e 2 respectivamente.

#include <stdio.h>
 
float media(float n1, float n2, float n3, char op) {
    
	if (op == 'A') {
		return (n1 + n2 + n3) / 3;
	} 
	else{
		return (n1 * 5 + n2 * 3 + n3 * 2) / 10;
	}
}
 
int main(){
    
	float nota1, nota2, nota3, resultado;
	char op;
    
	printf("Informe o cálculo desejado [A ou P]: ");
	scanf("%c", &op);
 
	printf("Informe primeira nota: ");
	scanf("%f", &nota1);
    
	printf("Informe a segunda nota: ");
	scanf("%f", &nota2);
    
	printf("Informe a terceira nota: ");
	scanf("%f", &nota3);
 
	resultado = media(nota1, nota2, nota3, op);
    
	printf("Media: %.1f", resultado);
    
	return 0;
}


5 - Escreva uma função que receba um símbolo e dois números. O símbolo representa a operação que se deseja efetuar com os números (soma, subtração, multiplicação ou divisão).

#include <stdio.h>
 
float calculo(float n1, float n2, char op) {
 
	switch(op){
		case '+': return n1 + n2; break;
		case '-': return n1 - n2; break;
		case '*': return n1 * n2; break;
		case '/': return n1 / n2; break;
	}
}
 
int main(){
 
	float num1, num2;
	char op;
    
	printf("Informe a operação desejada [+, -, * ou /]: ");
	scanf("%c", &op);
    
	printf("Informe o primeiro número: ");
	scanf("%f", &num1);
    
	printf("Informe o segundo número: ");
	scanf("%f", &num2);
 
	if(op == '/' && num2 == 0){
		printf("Não é possível a divisão por 0.");
	}
	else{
		printf("Resultado: %.1f", calculo(num1, num2, op));
	}
    
	return 0;
}


6 - Escreva uma função que receba como parâmetro um número inteiro e gere como saída n linhas com asteriscos, conforme o exemplo a seguir (para n = 5):

*
**
***
****
*****


#include <stdio.h>
 
void linhas(int num){
 
	int i, j;
    
	for(i = 1; i <= num; i++){
		for(j = 1; j <= i; j++){
			printf("*");
		}
		printf("\n");
	}
}
 
int main(){
 
	int num;
    
	printf("Informe o número de linhas desejadas: ");
	scanf("%d", &num);
 
	linhas(num);
    
	return 0;
}

9 - Escreva uma função que retorne a média dos valores contidos em um vetor com 5 elementos.

#include <stdio.h>
 
float media(float vet[5]){
    
    float resultado, soma = 0;
    
    for(int i = 0; i < 5; i++){
        soma += vet[i];
    }
    
    resultado = soma / 5;
    
    return resultado;
}
 
int main() {
    
    float vetor[5] = {1, 2, 3, 4, 5};
    
    printf("A média dos elementos do vetor é: %.1f", media(vetor));
 
    return 0;
}


10 - Escreva uma função que retorne a soma dos elementos de uma matriz 2 x 3.

#include <stdio.h>
 
int soma_matriz(int mat[2][3]){
    
    int soma = 0;
    
    for(int i = 0; i < 2; i++){
        for(int j = 0; j < 3; j++){
            soma += mat[i][j];
        }
    }
    
    return soma;
}
 
int main() {
    
    int matriz[2][3] = {{1, 2, 3}, {4, 5, 6}};
    
    printf("A soma é: %d", soma_matriz(matriz));
 
    return 0;
}
